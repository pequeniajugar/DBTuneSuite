employee_ssnum(
    ssnum,

    lat,

    hundreds1,
    hundreds2

    ssnum:          int     1 ... n          primary key (sorted ascending, unique)
    ssnumpermuted1: int     permutation of ssnum (unsorted)
    ssnumpermuted2: int     independent permutation of ssnum (unsorted)
    name:           text    name0, name1, ... (reused across rows)
    lat:            int     derived from hundreds1 (sorted ascending)
    long:           int     derived from hundreds1 (shuffled)
    hundreds1:      int     100 ... (100 + n/100 - 1), each repeated 100 times (sorted)
    hundreds2:      int     same value set as hundreds1 but shuffled

Data properties (as generated by employees.py):

- ssnum and hundreds1 are globally sorted ascending.
- lat and long are integer encodings of the same base value as hundreds1,
  but lat keeps the sorted order while long is shuffled.
- ssnumpermuted1 and ssnumpermuted2 are two independent permutations of ssnum.
- name and hundreds2 are shuffled within each batch.

create schema:

CREATE TABLE employee_ssnum (
    ssnum INT PRIMARY KEY,
    ssnumpermuted1 INT,
    ssnumpermuted2 INT,
    name VARCHAR(255),
    lat INT,
    long INT,
    hundreds1 INT,
    hundreds2 INT
);

-- Index definitions:
-- We define a primary key on ssnum, a non-clustered index on ssnumpermuted1,
-- and no index on ssnumpermuted2.

CREATE INDEX IF NOT EXISTS nc ON employee_ssnum(ssnumpermuted1);
